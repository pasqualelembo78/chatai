2.3 Trigger / Condizioni

Ogni milestone ha una condition che può essere:
  - type: "keyword" → match su testo utente (regex)
  - type: "topic" → match su topic estratti (analisi AI leggera)
  - type: "emotion" → match su emozione rilevata (es. "romance")
  - type: "message_count" → dopo N messaggi totali
  - type: "intimacy" → quando intimacy raggiunge X
  - type: "time" → ora del giorno / data specifica
  - type: "composite" → AND/OR di più condizioni

Ogni milestone può avere:
  - trigger_count: quante volte deve accadere
  - cooldown_messages: messaggi da aspettare prima di riattivarsi
  - one_shot: true = si attiva una volta sola
  - effect: cosa succede (delta relationship, unlock flag, unlock dialogo)
  - dialog_hint: testo da iniettare nel prompt per guidare l'AI
  - unlock_flag: flag da salvare su character_evolution


2.4 Evoluzione Engine — evolution_engine.py (nuovo file)

  from storage import get_evolution, update_evolution
  from emotion_engine import detect_emotion

  def evaluate_evolution(user_id, character_id, character, user_text, emotion):
      """
      Valuta ogni messaggio dell'utente contro i milestones
      del personaggio. Ritorna un dict con:
        - new_stage: str | None
        - unlocked: list di flag/dialoghi sbloccati
        - relationship_deltas: dict
        - trait_modifiers: dict
        - dialog_hints: list di stringhe per il prompt
      """
      evo = get_evolution(user_id, character_id)
      config = character.get("evolution", {})
      stages = config.get("stages", [])
      milestones = config.get("milestones", [])

      updates = {
          "new_stage": None,
          "unlocked": [],
          "relationship_deltas": {},
          "trait_modifiers": {},
          "dialog_hints": [],
      }

      # 1. Check stage progression
      for stage in stages:
          sid = stage["id"]
          if sid in evo["unlocked_stages"]:
              continue
          if evo["total_messages"] >= stage["min_messages"]:
              # Stage sbloccato!
              updates["new_stage"] = sid
              updates["unlocked"].extend(stage.get("unlocks", []))
              updates["trait_modifiers"].update(stage.get("trait_bonus", {}))
              # ... unlock ...

      # 2. Check milestones
      for m in milestones:
          if _check_condition(m, user_text, emotion, evo):
              if m.get("one_shot") and m["id"] in evo["flags"]:
                  continue
              # Attiva milestone
              updates["unlocked"].append(m.get("unlock_flag"))
              if "effect" in m:
                  updates["relationship_deltas"].update(m["effect"])
              if "dialog" in m:
                  updates["dialog_hints"].append(m["dialog"])
              evo["flags"][m["id"]] = True

      # 3. Update evolution in DB
      update_evolution(user_id, character_id, updates, evo)

      return updates

  def _check_condition(milestone, user_text, emotion, evo):
      cond = milestone.get("condition", {})
      ctype = cond.get("type")
      if ctype == "keyword":
          return any(kw in user_text.lower() for kw in cond.get("value", []))
      if ctype == "emotion":
          return emotion in cond.get("value", [])
      if ctype == "message_count":
          return evo["total_messages"] >= cond.get("value", 0)
      if ctype == "composite":
          op = cond.get("operator", "AND")
          results = [_check_condition(c, user_text, emotion, evo) for c in cond.get("conditions", [])]
          return all(results) if op == "AND" else any(results)
      return False

2.8 Storage — Nuove funzioni in storage.py

  def get_evolution(user_id, character_id):
      conn = get_conn()
      row = conn.execute(
          "SELECT * FROM character_evolution WHERE user_id=? AND character_id=?",
          (user_id, character_id)
      ).fetchone()
      conn.close()
      if row:
          return {
              "current_stage": row["current_stage"],
              "unlocked_stages": json.loads(row["unlocked_stages"]),
              "flags": json.loads(row["flags"]),
              "trait_modifiers": json.loads(row["trait_modifiers"]),
              "intimacy_peak": row["intimacy_peak"],
              "total_messages": row["total_messages"],
          }
      return {
          "current_stage": "base",
          "unlocked_stages": ["base"],
          "flags": {},
          "trait_modifiers": {},
          "intimacy_peak": 0,
          "total_messages": 0,
      }

  def update_evolution(user_id, character_id, updates, evo):
      conn = get_conn()
      if updates.get("new_stage"):
          evo["current_stage"] = updates["new_stage"]
          evo["unlocked_stages"].append(updates["new_stage"])
      if updates.get("unlocked"):
          evo["flags"].update({f: True for f in updates["unlocked"]})
      evo["total_messages"] = evo.get("total_messages", 0) + 1
      conn.execute("""
          INSERT OR REPLACE INTO character_evolution
          (user_id, character_id, current_stage, unlocked_stages,
           flags, trait_modifiers, intimacy_peak, total_messages, updated_at)
          VALUES (?, ?, ?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP)
      """, (
          user_id, character_id,
          evo["current_stage"],
          json.dumps(evo["unlocked_stages"]),
          json.dumps(evo["flags"]),
          json.dumps(evo.get("trait_modifiers", {})),
          evo.get("intimacy_peak", 0),
          evo["total_messages"],
      ))
      conn.commit()
      conn.close()
